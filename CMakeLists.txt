cmake_minimum_required(VERSION 3.16)
project(CCTVMonitor LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets)

# FFmpeg: folder berisi include/ dan lib/ (mis. build "shared" dari BtbN/FFmpeg-Builds)
set(FFMPEG_ROOT "" CACHE PATH "Folder FFmpeg (berisi include/ dan lib/)")
find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h HINTS ${FFMPEG_ROOT} PATH_SUFFIXES include)
set(FFMPEG_LIBS "")
foreach(c avformat avcodec swscale avutil)
  find_library(FFMPEG_${c}_LIB ${c} HINTS ${FFMPEG_ROOT} PATH_SUFFIXES lib)
  list(APPEND FFMPEG_LIBS ${FFMPEG_${c}_LIB})
endforeach()
if(NOT FFMPEG_INCLUDE_DIR OR NOT FFMPEG_avformat_LIB)
  message(FATAL_ERROR "FFmpeg tidak ditemukan. Set -DFFMPEG_ROOT=<folder ffmpeg>")
endif()

file(GLOB SRC CONFIGURE_DEPENDS src/*.cpp src/*.h)
if(WIN32)
  list(APPEND SRC ${CMAKE_CURRENT_SOURCE_DIR}/app.rc)
endif()
add_executable(CCTVMonitor WIN32 ${SRC})
target_include_directories(CCTVMonitor PRIVATE ${FFMPEG_INCLUDE_DIR})
target_link_libraries(CCTVMonitor PRIVATE Qt6::Widgets ${FFMPEG_LIBS})
if(MSVC)
  target_compile_options(CCTVMonitor PRIVATE /O2 /utf-8 /W3)
else()
  target_compile_options(CCTVMonitor PRIVATE -O2 -Wall -Wextra)
endif()
